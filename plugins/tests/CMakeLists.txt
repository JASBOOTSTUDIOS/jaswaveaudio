cmake_minimum_required(VERSION 3.25)
if(NOT DEFINED PROJECT_NAME)
  project(jaswave_plugin_tests VERSION 1.0.0 LANGUAGES C CXX)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/JasWavePlugins.cmake")

if(NOT TARGET jw_license_core)
  add_subdirectory("${JW_PLUGINS_ROOT}/shared-license" "${CMAKE_BINARY_DIR}/_shared-license")
endif()

# 1) Autoverificación numérica del DSP de todos los plugins (sin SDK, sin UI)
add_executable(jw_dsp_selftest dsp_selftest.cpp)
target_include_directories(jw_dsp_selftest PRIVATE "${JW_PLUGINS_ROOT}/shared-dsp" "${JW_PLUGINS_ROOT}")
if(MSVC)
  target_compile_options(jw_dsp_selftest PRIVATE /O2 /EHsc /W3 /fp:precise /utf-8)
  target_compile_definitions(jw_dsp_selftest PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()

# 2) Núcleo de licencias (RFC 8032 + licencias firmadas por Node con clave de prueba)
add_executable(jw_license_tests license/license_tests.cpp)
target_include_directories(jw_license_tests PRIVATE license "${CMAKE_BINARY_DIR}/_shared-license/gen")
target_link_libraries(jw_license_tests PRIVATE jw_license_core)
add_dependencies(jw_license_tests jw_license_pubkey)
target_compile_definitions(jw_license_tests PRIVATE JW_HAVE_PROD_KEY=1
  JW_PROD_PEM_PATH="${JW_PLUGINS_ROOT}/shared-license/keys/licencia_publica.pem" _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(jw_license_tests PRIVATE /utf-8 /EHsc /W3)
endif()

# 3) E2E del Manager (device-code + renovación) contra un backend falso en Node (run-e2e.mjs)
if(WIN32)
  add_executable(jw_license_e2e license/license_e2e.cpp)
  target_link_libraries(jw_license_e2e PRIVATE jw_license_testing)
  if(MSVC)
    target_compile_options(jw_license_e2e PRIVATE /utf-8 /EHsc /W3)
  endif()
endif()

enable_testing()
add_test(NAME dsp_selftest COMMAND jw_dsp_selftest)
add_test(NAME license_tests COMMAND jw_license_tests)
