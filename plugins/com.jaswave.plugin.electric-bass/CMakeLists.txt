# JasWave Electric Bass – VST3 instrumento independiente de bajo eléctrico (guía de ondas de cuerda)
cmake_minimum_required(VERSION 3.25)
project(jaswave_electric_bass VERSION 1.0.0 LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/JasWavePlugins.cmake")
jw_setup_shared()
jaswave_add_vst3(TARGET JasWaveElectricBass SOURCES source/electric_bass_plugin.cpp INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/source")

# Prueba de render nativa (solo el DSP, sin SDK ni UI): afinación en cents, pico, RMS, CPU y WAV
add_executable(JasWaveElectricBass_rendertest test/render_test.cpp)
target_include_directories(JasWaveElectricBass_rendertest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/source" "${JW_PLUGINS_ROOT}/shared-dsp")
if(MSVC)
  target_compile_options(JasWaveElectricBass_rendertest PRIVATE /O2 /EHsc /W3 /fp:precise /utf-8)
  target_compile_definitions(JasWaveElectricBass_rendertest PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
set_target_properties(JasWaveElectricBass_rendertest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools/$<CONFIG>")
